cmake_minimum_required(VERSION 3.16)
project(BibliotecaAcademica VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Instrumentar todo el C++ con ASan/UBSan" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ENABLE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers requieren GCC o Clang")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(libro backend/src/Libro.cpp)
target_include_directories(libro PUBLIC backend/include)
add_library(bplus backend/src/BPlusTree.cpp)
target_link_libraries(bplus PUBLIC libro)
add_library(json_repository backend/src/JsonBookRepository.cpp)
target_link_libraries(json_repository PUBLIC libro)
target_include_directories(json_repository SYSTEM PRIVATE third_party)
add_library(catalog_service backend/src/CatalogService.cpp)
target_link_libraries(catalog_service PUBLIC bplus)
add_library(console_ui backend/src/ConsoleUI.cpp)
target_link_libraries(console_ui PUBLIC catalog_service)
add_executable(biblioteca backend/apps/main.cpp)
target_link_libraries(biblioteca PRIVATE console_ui json_repository)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_bplus backend/tests/test_bplus.cpp)
  target_link_libraries(test_bplus PRIVATE bplus)
  add_test(NAME bplus_oraculo COMMAND test_bplus)
  add_executable(test_catalog backend/tests/test_catalog.cpp)
  target_link_libraries(test_catalog PRIVATE catalog_service json_repository console_ui)
  target_compile_definitions(test_catalog PRIVATE SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  add_test(NAME catalogo_json_consola COMMAND test_catalog)
  add_test(NAME demo_reinicio COMMAND bash ${CMAKE_SOURCE_DIR}/scripts/demo_entregable2.sh $<TARGET_FILE:biblioteca>)
  # ASan tardó 177 s en WSL; margen para equipos más lentos, sin reducir casos.
  set_tests_properties(bplus_oraculo catalogo_json_consola demo_reinicio PROPERTIES TIMEOUT 600)
endif()
